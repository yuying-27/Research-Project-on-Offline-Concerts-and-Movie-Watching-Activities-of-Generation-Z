# ---------- 数据准备 ----------
df = pd.read_excel('C:/Users/ui_07/Documents/OneDrive/桌面/问卷量表数据已编码(2).xlsx')
df_clean = df.iloc[:, :-1].reset_index(drop=True)
for col in df_clean.columns:
    df_clean[col] = pd.to_numeric(df_clean[col], errors='coerce')

indicators = pd.DataFrame()
indicators['艺人忠诚度'] = df_clean.iloc[:, 0]
indicators['消费动机'] = df_clean.iloc[:, 1]
indicators['单次消费金额'] = (df_clean.iloc[:, 4] + df_clean.iloc[:, 6]) / 2
indicators['观演频率'] = df_clean.iloc[:, 3]
indicators['价格敏感度'] = 6 - df_clean.iloc[:, 5]
indicators['体验诉求'] = (df_clean.iloc[:, 7] + df_clean.iloc[:, 8] + df_clean.iloc[:, 9]) / 3
indicators['氛围感受'] = df_clean.iloc[:, 2]

scaler = StandardScaler()
X_scaled = scaler.fit_transform(indicators)

kmeans = KMeans(n_clusters=3, random_state=42, n_init=10, max_iter=300)
cluster_labels = kmeans.fit_predict(X_scaled)
indicators['聚类标签'] = cluster_labels

cluster_names = {0: '边缘观望型', 1: '狂热核心型', 2: '价格敏感型'}
indicators['类别名称'] = indicators['聚类标签'].map(cluster_names)

colors = ['#e74c3c', '#3498db', '#2ecc71']
cluster_names_list = ['边缘观望型', '狂热核心型', '价格敏感型']
markers = ['o', 's', '^']

# ---------- 图1: PCA散点图（带聚类中心和凸包） ----------
fig, ax = plt.subplots(figsize=(12, 9))

for i in range(3):
    mask = cluster_labels == i
    ax.scatter(X_pca[mask, 0], X_pca[mask, 1], 
               c=colors[i], marker=markers[i], s=60, alpha=0.6, 
               edgecolors='white', linewidth=0.5,
               label=f'类别{i}: {cluster_names_list[i]} (n={sum(mask)})')

    # 聚类中心
    ax.scatter(centers_pca[i, 0], centers_pca[i, 1], 
               c=colors[i], marker='*', s=800, edgecolors='black', 
               linewidth=2, zorder=10)
    ax.annotate(f'中心{i}', 
                xy=(centers_pca[i, 0], centers_pca[i, 1]),
                xytext=(centers_pca[i, 0] + 0.3, centers_pca[i, 1] + 0.3),
                fontsize=12, fontweight='bold', color=colors[i],
                arrowprops=dict(arrowstyle='->', color=colors[i], lw=1.5))

    # 凸包轮廓
    points = X_pca[mask]
    if len(points) > 3:
        try:
            hull = ConvexHull(points)
            hull_points = points[hull.vertices]
            hull_points = np.vstack([hull_points, hull_points[0]])
            ax.plot(hull_points[:, 0], hull_points[:, 1], '--', 
                    color=colors[i], alpha=0.4, linewidth=1.5)
        except:
            pass

ax.set_xlabel(f'PC1 ({pca.explained_variance_ratio_[0]*100:.1f}%)', fontsize=13)
ax.set_ylabel(f'PC2 ({pca.explained_variance_ratio_[1]*100:.1f}%)', fontsize=13)
ax.set_title('K-Means聚类结果散点图 (PCA降维)\n★ 标记为聚类中心', 
             fontsize=15, fontweight='bold', pad=15)
ax.legend(loc='best', fontsize=11, framealpha=0.9)
ax.grid(True, alpha=0.3)
ax.axhline(y=0, color='gray', linestyle='-', alpha=0.3)
ax.axvline(x=0, color='gray', linestyle='-', alpha=0.3)
plt.tight_layout()
plt.savefig('聚类散点图_PCA.png', dpi=300, bbox_inches='tight')
plt.show()
