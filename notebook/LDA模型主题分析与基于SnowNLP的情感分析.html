def clean_text(text):
    text = str(text)

import pandas as pd
import matplotlib.pyplot as plt
from gensim import corpora
from gensim.models.ldamodel import LdaModel
from gensim.models import CoherenceModel
from gensim.models import Phrases 
import warnings

# 忽略 Gensim 的弃用警告，保持控制台整洁
warnings.filterwarnings("ignore", category=DeprecationWarning)

plt.rcParams['font.sans-serif'] = ['SimHei']  # Windows 用户使用黑体
# plt.rcParams['font.sans-serif'] = ['Arial Unicode MS'] 
plt.rcParams['axes.unicode_minus'] = False

# 步骤 1: 读取真实数据与预处理

print("正在读取数据...")
file_path = r"C:\Users\20181\5.1_cleaned_data.xlsx"
df = pd.read_excel(file_path)

texts = []
for text in df['tokens']:
    if pd.isna(text):       
        texts.append([])
    else:
        # 过滤长度<2的词，同时过滤掉自定义的无意义停用词
        word_list = [
            word.strip()
            for word in str(text).split()
            if len(word.strip()) >= 2 and word.strip() not in stop_words
        ]
        texts.append(word_list)

print("读取完成！共有", len(texts), "条文本")

# 步骤 1.5: 挖掘多字词组 (N-grams) - 解决全都是两字词的问题

print("\n正在挖掘多字长词(N-grams)...")

bigram_model = Phrases(texts, min_count=3, threshold=5)

# 训练 Trigram (三元) 模型：在二元的基础上继续合并，可以生成更长的词
trigram_model = Phrases(bigram_model[texts], min_count=2, threshold=5)

# 将长词模型应用到我们的文本数据中
texts_with_ngrams = list(trigram_model[bigram_model[texts]])

# 去除 Gensim 自动添加的下划线 '_'，让 "官方_平台" 变成 "官方平台" (变身四字长词)
final_texts = []
for doc in texts_with_ngrams:
    final_texts.append([word.replace('_', '') for word in doc])

print("\n正在构建词典与语料库...")

dictionary = corpora.Dictionary(final_texts)

dictionary.filter_extremes(no_below=3, no_above=0.8)

# 将文本转化为词频矩阵
corpus = [dictionary.doc2bow(text) for text in final_texts]

# 步骤 3: 确定主题数(K) - 计算困惑度与一致性

def compute_metrics(dictionary, corpus, texts, limit, start=5, step=1):
    coherence_values = []
    perplexity_values = []
    model_list = []
    
    print("\n开始计算不同 K 值的模型评估指标 (可能需要几分钟)...")
    for num_topics in range(start, limit, step):
        model = LdaModel(corpus=corpus, num_topics=num_topics, id2word=dictionary, 
                         passes=15, random_state=42) 
        model_list.append(model)
        
        perplexity_values.append(model.log_perplexity(corpus))
        
        # 注意这里也要用 final_texts 计算一致性
        coherencemodel = CoherenceModel(model=model, texts=texts, dictionary=dictionary, coherence='c_v')
        coherence_score = coherencemodel.get_coherence()
        coherence_values.append(coherence_score)
        
        print(f"K={num_topics} 训练完成 -> 一致性 (Coherence): {coherence_score:.4f}")
        
    return model_list, coherence_values, perplexity_values

start_k, limit_k, step_k = 5, 11, 1
model_list, coherence_values, perplexity_values = compute_metrics(dictionary, corpus, final_texts, limit_k, start_k, step_k)

# 画折线图
x = range(start_k, limit_k, step_k)
fig, ax1 = plt.subplots(figsize=(10, 5))

color = 'tab:blue'
ax1.set_xlabel('主题数 (K)')
ax1.set_ylabel('一致性得分 (Coherence)', color=color)
ax1.plot(x, coherence_values, marker='o', color=color, label='Coherence (越高越好)')
ax1.tick_params(axis='y', labelcolor=color)

ax2 = ax1.twinx()  
color = 'tab:red'
ax2.set_ylabel('困惑度 (Log Perplexity)', color=color)  
ax2.plot(x, perplexity_values, marker='s', linestyle='--', color=color, label='Perplexity (越低越好)')
ax2.tick_params(axis='y', labelcolor=color)

fig.tight_layout()
plt.title('不同 K 值的 LDA 模型表现 (优先看蓝色一致性最高点)')
plt.show()

# 步骤 4: 训练最终模型与结果解析

optimal_k = 6
print(f"\n--- 选取主题数 K={optimal_k} 提取特征 ---")

best_lda_model = model_list[optimal_k - start_k]

print("\n各主题 Top 15 关键词 (已包含多字长词)：")
topics_keywords = {}
for idx, topic in best_lda_model.print_topics(-1, num_words=15):
    words = [word.split('*')[1].replace('"', '').strip() for word in topic.split('+')]
    topics_keywords[f'主题 {idx+1}'] = words
    print(f"主题 {idx+1}: {', '.join(words)}")

# 步骤 5: 将主题归属回填并保存

print("\n正在计算每条评论的归属主题...")
dominant_topics = []
for i, bow in enumerate(corpus):
    if not bow:
        dominant_topics.append("无有效文本")
        continue
    topic_probs = best_lda_model.get_document_topics(bow)
    dominant_topic = max(topic_probs, key=lambda x: x[1])[0]
    dominant_topics.append(f"主题 {dominant_topic + 1}")

df['Dominant_Topic'] = dominant_topics

topic_counts = df[df['Dominant_Topic'] != '无有效文本']['Dominant_Topic'].value_counts()
plt.figure(figsize=(8, 8))
plt.pie(topic_counts, labels=topic_counts.index, autopct='%1.1f%%', startangle=140, colors=plt.cm.Pastel1.colors)
plt.title('各 LDA 主题在全量数据中的占比')
plt.show()

output_file = '5_2_lda_results_with_longwords.csv'
df.to_csv(output_file, index=False, encoding='utf-8-sig')
print(f"\n处理完成！带有主题分类的结果已保存至: {output_file}")

import pandas as pd

# 1. 情感打分与分类

def get_sentiment_score(text):
    try:
        # SnowNLP 输出 0~1 之间的得分，越接近 1 越积极
        s = SnowNLP(str(text))
        return s.sentiments
    except:
        return 0.5  # 如果遇到空值或无法处理的文本，默认给中性分

print("正在计算情感得分，这可能需要一点时间...")

df['sentiment_score'] = df['评论'].apply(get_sentiment_score)

# 设定阈值进行情绪分类
def classify_emotion(score):
    if score > 0.6:
        return '积极'
    elif score < 0.4:
        return '消极'
    else:
        return '中性'

df['emotion_label'] = df['sentiment_score'].apply(classify_emotion)

# 2. 整体情感分布（饼图）

plt.figure(figsize=(8, 6))
emotion_counts = df['emotion_label'].value_counts()
colors = ['#ff9999', '#66b3ff', '#99ff99'] # 设定消极、积极、中性的颜色

plt.pie(emotion_counts, labels=emotion_counts.index, autopct='%1.1f%%', 
        startangle=140, colors=colors, shadow=True, textprops={'fontsize': 12})
plt.title('全量数据整体情感分布', fontsize=15)
plt.show()

# 3. 交叉分析：不同主题的情感表现

# 计算每个主题的平均情感得分和负面评论占比
topic_sentiment_stats = df.groupby('Dominant_Topic').agg(
    avg_score=('sentiment_score', 'mean'),

# 计算负面占比 (%)
topic_sentiment_stats['negative_ratio'] = (topic_sentiment_stats['negative_count'] / topic_sentiment_stats['total_count']) * 100

print("\n各主题情感统计表：")
print(topic_sentiment_stats)

top50_idx = mean_scores.argsort()[::-1][:50]

# 第三步：构建关键词共现矩阵

print('\n正在构建共现矩阵...')

top50_set = set(top50_words)
co_occurrence = defaultdict(int)

for tokens in df['filtered_tokens']:
    filtered = [w for w in tokens if w in top50_set]
    if len(filtered) < 2:
        continue
    for w1, w2 in itertools.combinations(sorted(set(filtered)), 2):
        co_occurrence[(w1, w2)] += 1

print(f'共现对数量：{len(co_occurrence)}')

top30 = top50_words[:30]
co_matrix = pd.DataFrame(0, index=top30, columns=top30)
for (w1, w2), cnt in co_occurrence.items():

        co_matrix.loc[w1, w2] = cnt
        co_matrix.loc[w2, w1] = cnt
co_matrix.to_csv('cooccurrence_matrix_top30.csv', encoding='utf-8-sig')

# 第四步：NetworkX共现网络 + 度中心性
