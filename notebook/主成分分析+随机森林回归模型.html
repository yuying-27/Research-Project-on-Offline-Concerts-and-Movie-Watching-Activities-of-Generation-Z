# Varimax旋转函数
def varimax(Phi, gamma=1.0, q=20, tol=1e-6):
    p, k = Phi.shape
    R = np.eye(k)
    d = 0
    for i in range(q):
        d_old = d
        Lambda = np.dot(Phi, R)
        u, s, vh = np.linalg.svd(
            np.dot(Phi.T, np.asarray(Lambda) ** 3 - (gamma / p) * np.dot(Lambda, np.diag(np.diag(np.dot(Lambda.T, Lambda)))))
        )
        R = np.dot(u, vh)
        d = np.sum(s)
        if d_old != 0 and d / d_old < 1 + tol:
            break
    return np.dot(Phi, R), R

# Cell 3: 因子载荷分析（PCA + Varimax旋转）

# 关键修正：用PCA提取主成分作为未旋转因子（SPSS默认做法）
pca_n = PCA(n_components=n_factors)
pca_n.fit(X_std)

# PCA载荷矩阵 = 特征向量 × √特征值
# 这样保证：每列平方和 = 特征值，每行平方和 = 共同度
loadings_unrotated = pca_n.components_.T * np.sqrt(pca_n.explained_variance_)

# 验证未旋转时共同度之和 = 特征值之和
communalities_unrot = (loadings_unrotated ** 2).sum(axis=1)
print(f'\n[验证] 未旋转共同度合计: {communalities_unrot.sum():.3f}')
print(f'[验证] 前{n_factors}个特征值合计: {eigenvalues[:n_factors].sum():.3f}')
print(f'[验证] 两者应相等: {"是" if abs(communalities_unrot.sum() - eigenvalues[:n_factors].sum()) < 0.01 else "否"}')

# 应用Varimax旋转
loadings_rotated, rotation_matrix = varimax(loadings_unrotated)

# 符号翻转：使每个因子中载荷绝对值最大的项为正（便于解释）
for j in range(loadings_rotated.shape[1]):
    max_idx = np.argmax(np.abs(loadings_rotated[:, j]))
    if loadings_rotated[max_idx, j] < 0:
        loadings_rotated[:, j] = -loadings_rotated[:, j]

# 按方差解释率降序排列因子

order = np.argsort(-ss_loadings_init)

# 构建载荷矩阵DataFrame

communalities = (loadings_rotated ** 2).sum(axis=1)

# 计算旋转后的方差解释率
# Varimax正交旋转保持总方差不变：旋转后载荷平方和之和 = 旋转前特征值之和

# Cell 4: 计算因子得分（回归法）

# 因子得分系数 = 载荷矩阵的伪逆 × 原始变量
# 使用回归法计算因子得分

    elif any(k in items_text for k in ['加强秩序管控', '明确处罚措施', '现场安全考量']):
        return '秩序安全期望'
    return '其他维度'

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.ensemble import RandomForestRegressor
from sklearn.model_selection import train_test_split, cross_val_score
from sklearn.inspection import PartialDependenceDisplay
from sklearn.metrics import r2_score, mean_squared_error, mean_absolute_error
import warnings
warnings.filterwarnings('ignore')

plt.rcParams['font.sans-serif'] = ['SimHei', 'Microsoft YaHei', 'Noto Sans CJK SC']
plt.rcParams['axes.unicode_minus'] = False

# Cell 1: 数据读取与预处理

file_path = 'C:/Users/20181/Desktop/第五六章代码及数据合集/问卷数据/问卷量表数据已编码.xlsx'
df = pd.read_excel(file_path)

labels = [
    '喜欢艺人', '朋友娱乐', '现场氛围', '释放压力',
    '座位影响价值', '定价不合理', '愿付高价', '购票选位满意度',
    '加强秩序管控', '明确处罚措施', '现场安全考量'
]
df.columns = labels

# 负向题反向编码 + 重命名列（与EFA保持一致）
df['定价不合理'] = 6 - df['定价不合理']
df.rename(columns={'定价不合理': '定价合理性'}, inplace=True)

# 定义X和Y
y = df['愿付高价'].values
X = df.drop(columns=['愿付高价']).values
feature_names = df.drop(columns=['愿付高价']).columns.tolist()

print(f'[数据概况] 样本量: {len(y)}, 自变量数: {len(feature_names)}')
print(f'因变量Y: 愿付高价')
print(f'自变量X: {feature_names}')
print(f'\n因变量Y分布:')
for v in sorted(df['愿付高价'].unique()):
    print(f'  {v}分: {(df["愿付高价"] == v).sum()}人 ({(df["愿付高价"] == v).mean()*100:.1f}%)')

# Cell 2: 划分训练集与测试集 + 训练随机森林模型

X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.3, random_state=42
)

print(f'[数据划分] 总样本: {len(y)}条')
print(f'  训练集: {len(y_train)}条（{len(y_train)/len(y)*100:.0f}%）')
print(f'  测试集: {len(y_test)}条（{len(y_test)/len(y)*100:.0f}%）')

rf = RandomForestRegressor(
    n_estimators=500,
    max_depth=10,
    min_samples_leaf=5,
    random_state=42,
    n_jobs=-1
)
rf.fit(X_train, y_train)

y_pred_train = rf.predict(X_train)
y_pred_test = rf.predict(X_test)

print(f'训练集 R² = {r2_score(y_train, y_pred_train):.4f}')
print(f'测试集 R² = {r2_score(y_test, y_pred_test):.4f}')
print(f'测试集 RMSE = {np.sqrt(mean_squared_error(y_test, y_pred_test)):.4f}')
print(f'测试集 MAE = {mean_absolute_error(y_test, y_pred_test):.4f}')

# 5折交叉验证
cv_scores = cross_val_score(rf, X, y, cv=5, scoring='r2')
print(f'\n5折交叉验证 R² = {cv_scores.mean():.4f} ± {cv_scores.std():.4f}')
